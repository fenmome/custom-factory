## 一、技术选型总览

| 层 | 选型 | 版本 | 理由 |
|----|------|------|------|
| **前端框架** | Vue 3 | ^3.4 | 生态成熟、中文社区强、模板多 |
| **前端 UI** | Element Plus / Ant Design Vue | 最新 | 后台管理系统标准组件库 |
| **前端状态** | Pinia | ^2.x | Vue 3 官方推荐 |
| **前端路由** | Vue Router | ^4.x | |
| **构建工具** | Vite | ^5.x | 快速热更新 |
| **后端框架** | FastAPI | ^0.115 | 异步、自动文档、Python 生态 |

| **反向代理** | Nginx | latest | 前端静态 + API 转发 |

## 四、Docker 部署

### 4.1 docker-compose.yml（生产）

```yaml
version: "3.8"

services:
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./frontend/dist:/usr/share/nginx/html:ro
      - ./nginx/conf.d:/etc/nginx/conf.d:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro
    depends_on:
      - backend
    networks:
      - cf-network

  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    volumes:
      - ./frontend/dist:/app/dist
    networks:
      - cf-network

  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    environment:
      - DATABASE_URL=mysql+pymysql://cf_user:${DB_PASSWORD}@mysql:3306/custom_factory
      - REDIS_URL=redis://redis:6379/0
      - STORAGE_ROOT=/app/data/images
      - SECRET_KEY=${SECRET_KEY}
      - ENVIRONMENT=production
    volumes:
      - ./data/images:/app/data/images
      - ./backend/app/resources:/app/app/resources:ro
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - cf-network
    restart: unless-stopped

  mysql:
    image: mysql:8.0
    environment:
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD}
      MYSQL_DATABASE: custom_factory
      MYSQL_USER: cf_user
      MYSQL_PASSWORD: ${DB_PASSWORD}
    volumes:
      - ./data/mysql:/var/lib/mysql
      - ./scripts/init-db.sh:/docker-entrypoint-initdb.d/init.sh
    ports:
      - "3306:3306"
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - cf-network

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    volumes:
      - ./data/redis:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - cf-network

networks:
  cf-network:
```

## 五、后端设计

### 5.1 FastAPI 应用入口

```python
# app/main.py
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from app.config import settings
from app.api.v1 import batches, products, reviews, qa, guard, auth, workspace


def create_app() -> FastAPI:
    app = FastAPI(
        title="Custom Factory API",
        version="1.0.0",
        docs_url="/api/docs" if settings.ENVIRONMENT != "production" else None,
        redoc_url=None,
    )

    app.add_middleware(
        CORSMiddleware,
        allow_origins=settings.CORS_ORIGINS,
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    app.include_router(auth.router, prefix="/api/v1/auth", tags=["认证"])
    app.include_router(workspace.router, prefix="/api/v1/workspaces", tags=["工作空间"])
    app.include_router(batches.router, prefix="/api/v1/batches", tags=["批次"])
    app.include_router(products.router, prefix="/api/v1/products", tags=["商品"])
    app.include_router(reviews.router, prefix="/api/v1/reviews", tags=["复审"])
    app.include_router(qa.router, prefix="/api/v1/qa", tags=["质检"])
    app.include_router(guard.router, prefix="/api/v1/guard", tags=["生产保护"])
    # Runner API（内部服务间调用）
    app.include_router(
        runner.router, prefix="/internal/runner", tags=["Runner"]
    )

    return app


app = create_app()
```

### 5.3 数据库模型（核心表 MySQL DDL）

```sql
-- 工作空间与用户
CREATE TABLE workspaces (
    id CHAR(36) PRIMARY KEY,
    display_name VARCHAR(255) NOT NULL,
    state ENUM('ACTIVE','SUSPENDED','ARCHIVED') DEFAULT 'ACTIVE',
    created_at TIMESTAMP(6) NOT NULL,
    updated_at TIMESTAMP(6) NOT NULL,
    INDEX idx_state (state)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE users (
    id CHAR(36) PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    display_name VARCHAR(255) NOT NULL,
    state ENUM('ACTIVE','DEACTIVATED') DEFAULT 'ACTIVE',
    created_at TIMESTAMP(6) NOT NULL,
    updated_at TIMESTAMP(6) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE workspace_memberships (
    id CHAR(36) PRIMARY KEY,
    workspace_id CHAR(36) NOT NULL,
    user_id CHAR(36) NOT NULL,
    role ENUM('OWNER','MEMBER','REVIEWER') NOT NULL,
    status ENUM('ACTIVE','REVOKED') DEFAULT 'ACTIVE',
    joined_at TIMESTAMP(6) NOT NULL,
    revoked_at TIMESTAMP(6) NULL,
    UNIQUE KEY uk_ws_user (workspace_id, user_id),
    FOREIGN KEY (workspace_id) REFERENCES workspaces(id),
    FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 批次与商品
CREATE TABLE batches (
    id CHAR(36) PRIMARY KEY,
    workspace_id CHAR(36) NOT NULL,
    requested_quantity INT NOT NULL,
    image_generation_route ENUM('API','CODEX') NOT NULL,
    control_state VARCHAR(32) DEFAULT 'ACTIVE',
    request_key VARCHAR(128) NOT NULL,
    input_snapshot_hash CHAR(64) NOT NULL,
    created_by_user_id CHAR(36) NOT NULL,
    created_at TIMESTAMP(6) NOT NULL,
    updated_at TIMESTAMP(6) NOT NULL,
    INDEX idx_ws_created (workspace_id, created_at),
    UNIQUE KEY uk_request (workspace_id, request_key),
    FOREIGN KEY (workspace_id) REFERENCES workspaces(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE products (
    id CHAR(36) PRIMARY KEY,
    workspace_id CHAR(36) NOT NULL,
    batch_id CHAR(36) NOT NULL,
    sku_namespace CHAR(36) NOT NULL,       -- product_type_id
    internal_sku VARCHAR(64) NOT NULL,      -- DZUVDTF000001
    allocated_prefix VARCHAR(32) NOT NULL,
    allocated_sequence INT NOT NULL,
    state ENUM('ACTIVE','CANCELLED','ARCHIVED') DEFAULT 'ACTIVE',
    created_at TIMESTAMP(6) NOT NULL,
    updated_at TIMESTAMP(6) NOT NULL,
    UNIQUE KEY uk_sku (internal_sku),
    INDEX idx_batch (batch_id),
    FOREIGN KEY (batch_id) REFERENCES batches(id),
    FOREIGN KEY (workspace_id) REFERENCES workspaces(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE product_revisions (
    id CHAR(36) PRIMARY KEY,
    workspace_id CHAR(36) NOT NULL,
    product_id CHAR(36) NOT NULL,
    batch_id CHAR(36) NOT NULL,
    revision_number INT NOT NULL DEFAULT 1,
    concept_id CHAR(36) NOT NULL,
    status ENUM('DRAFT','FROZEN','READY_FOR_PUBLISH') DEFAULT 'DRAFT',
    image_policy_id VARCHAR(64),
    image_policy_version VARCHAR(32),
    image_policy_content_hash CHAR(64),
    ready_gate_id CHAR(36) NULL,
    created_at TIMESTAMP(6) NOT NULL,
    updated_at TIMESTAMP(6) NOT NULL,
```

### 5.4 认证方案

```python
# app/core/security.py
from datetime import datetime, timedelta, timezone
from jose import JWTError, jwt
from passlib.context import CryptContext
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
security = HTTPBearer()


def create_access_token(user_id: str, workspace_id: str) -> str:
    payload = {
        "sub": user_id,
        "ws": workspace_id,
        "exp": datetime.now(timezone.utc) + timedelta(hours=8),
    }
    return jwt.encode(payload, settings.SECRET_KEY, algorithm="HS256")


def verify_token(token: str) -> dict:
    try:
        return jwt.decode(token, settings.SECRET_KEY, algorithms=["HS256"])
    except JWTError:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "Invalid token")


async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db),
) -> tuple[str, str]:
    payload = verify_token(credentials.credentials)
    user_id, workspace_id = payload["sub"], payload["ws"]
    # 验证成员关系
    membership = db.scalar(
        select(WorkspaceMembership)
        .where(
            WorkspaceMembership.user_id == user_id,
            WorkspaceMembership.workspace_id == workspace_id,
            WorkspaceMembership.status == "ACTIVE",
        )
    )
    if not membership:
        raise HTTPException(403, "Invalid workspace membership")
    return user_id, workspace_id
```

```typescript
import axios from 'axios'

client.interceptors.response.use(
  (response) => response.data,

)
```

### 6.5 状态管理（Pinia）

```typescript
// stores/auth.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import client from '@/api/client'

export const useAuthStore = defineStore('auth', () => {
  const token = ref(localStorage.getItem('token') || '')
  const user = ref(null)
  const workspaceId = ref(localStorage.getItem('ws') || '')

  async function login(email: string, password: string) {
    const data = await client.post('/auth/login', { email, password })
    token.value = data.token
    workspaceId.value = data.workspace_id
    localStorage.setItem('token', data.token)
    localStorage.setItem('ws', data.workspace_id)
  }

  function logout() {
    token.value = ''
    workspaceId.value = ''
    localStorage.clear()
  }

  return { token, user, workspaceId, login, logout }
})
```
